# TP-1: Cleaning Data — OOP & Design Patterns

```python
import time
from abc import ABC, abstractmethod
from functools import wraps
import pandas as pd


# Exercise 3 — Decorators
def log_action(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        print(f"Starting {func.__name__}")
        try:
            return func(*args, **kwargs)
        finally:
            print(f"Finished {func.__name__}")
    return wrapper


def log_time(func):
    @wraps(func)
    def wrapper(*args, **kwargs):
        start = time.perf_counter()
        try:
            return func(*args, **kwargs)
        finally:
            print(f"Time: {time.perf_counter() - start:.4f}s")
    return wrapper


# Exercise 1 — CSVReader
class CSVReader:
    def __init__(self, file_path):
        self.file_path = file_path

    @log_action
    @log_time
    def read(self):
        return pd.read_csv(self.file_path)

    def preview(self, n=5):
        print(self.read().head(n))


# Exercise 2 — Strategy Pattern
class MissingValueStrategy(ABC):
    @abstractmethod
    def handle(self, df):
        pass


class DropMissing(MissingValueStrategy):
    def handle(self, df):
        return df.dropna()


class FillMean(MissingValueStrategy):
    def handle(self, df):
        return df.fillna(df.mean(numeric_only=True))


class FillMode(MissingValueStrategy):
    def handle(self, df):
        df = df.copy()
        for column in df.columns:
            mode = df[column].mode()
            if not mode.empty:
                df[column] = df[column].fillna(mode.iloc[0])
        return df


class DataCleaner:
    def __init__(self, strategy):
        self.strategy = strategy

    def clean(self, df):
        return self.strategy.handle(df)


# Exercise 4 — Factory Pattern
class DataTransform(ABC):
    @abstractmethod
    def apply(self, df):
        pass


class NormalizeColumns(DataTransform):
    def apply(self, df):
        df = df.copy()
        numbers = df.select_dtypes(include="number")
        scale = (numbers.max() - numbers.min()).replace(0, 1)
        df[numbers.columns] = (numbers - numbers.min()) / scale
        return df


class RemoveDuplicates(DataTransform):
    def apply(self, df):
        return df.drop_duplicates()


class StandardizeText(DataTransform):
    def apply(self, df):
        df = df.copy()
        for col in df.select_dtypes(include=["object", "string"]).columns:
            df[col] = df[col].map(
                lambda x: x.strip().lower() if isinstance(x, str) else x
            )
        return df


class TransformFactory:
    @staticmethod
    def create(name):
        transforms = {
            "normalize": NormalizeColumns,
            "remove_duplicates": RemoveDuplicates,
            "standardize_text": StandardizeText,
        }
        return transforms[name]()


# Exercise 5 — Template Method Pattern
class DataPipeline(ABC):
    def run(self):
        df = self.load()
        df = self.clean(df)
        df = self.transform(df)
        self.save(df)
        return df

    @abstractmethod
    def load(self):
        pass

    @abstractmethod
    def clean(self, df):
        pass

    @abstractmethod
    def transform(self, df):
        pass

    @abstractmethod
    def save(self, df):
        pass


class CSVDataPipeline(DataPipeline):
    def __init__(self, input_path, output_path, strategy, transforms):
        self.reader = CSVReader(input_path)
        self.cleaner = DataCleaner(strategy)
        self.transforms = [TransformFactory.create(name) for name in transforms]
        self.output_path = output_path

    def load(self):
        return self.reader.read()

    def clean(self, df):
        return self.cleaner.clean(df)

    def transform(self, df):
        for transform in self.transforms:
            df = transform.apply(df)
        return df

    def save(self, df):
        df.to_csv(self.output_path, index=False)


# Example
pipeline = CSVDataPipeline(
    "input.csv", "cleaned.csv", FillMean(),
    ["remove_duplicates", "standardize_text", "normalize"]
)
pipeline.run()
```
